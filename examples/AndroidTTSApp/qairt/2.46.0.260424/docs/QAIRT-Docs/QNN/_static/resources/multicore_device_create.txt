   QnnDevice_CoreInfo_t coreInfo[2];
   coreInfo[0] = platformInfo->v1.hwDevices[deviceId].v1.cores[A];
   coreInfo[1] = platformInfo->v1.hwDevices[deviceId].v1.cores[B];   
   
   QnnDevice_HardwareDeviceInfo_t hwDeviceInfo;
   hwDeviceInfo.v1.deviceId      = platformInfo->v1.hwDevices[deviceId].v1.deviceId;
   hwDeviceInfo.v1.deviceType  = platformInfo->v1.hwDevices[deviceId].v1.deviceType;
   hwDeviceInfo.v1.numCores    = 2;
   hwDeviceInfo.v1.cores           = &coreInfo;
    

   QnnDevice_PlatformInfo_t testPlatformInfo;
   testPlatformInfo.v1.numHwDevices = 1;
   testPlatformInfo.v1.hwDevices        = &hwDeviceInfo;

   QnnDevice_Config_t deviceConfig;
   deviceConfig.option           = QNN_DEVICE_CONFIG_OPTION_PLATFORM_INFO;
   deviceConfig.hardwareInfo = &testPlatformInfo;
   QnnDevice_Config_t* deviceConfigs2cores[] = {&deviceConfig, nullptr};
   
   Qnn_DeviceHandle_t multicoreDeviceHandle;
   Qnn_LogHandle_t logHandle;
